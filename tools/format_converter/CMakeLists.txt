cmake_minimum_required(VERSION 3.22)
project(easypath_converter_ui LANGUAGES CXX)
find_package(FLTK 1.4 CONFIG REQUIRED)
find_package(Threads REQUIRED)
add_library(converter_ui SHARED desktop.cpp)
target_compile_features(converter_ui PRIVATE cxx_std_17)
target_link_libraries(converter_ui PRIVATE fltk::fltk Threads::Threads)
if(WIN32)
  set_target_properties(converter_ui PROPERTIES PREFIX "")
endif()
